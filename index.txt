async function getYoungestAndOldest() {
  const users = await User.find();

  if (users.length === 0) {
    return null;
  }

  const youngest = users.reduce((min, user) => {
    return user.age < min.age ? user : min;
  });

  const oldest = users.reduce((max, user) => {
    return user.age > max.age ? user : max;
  });

  return {
    youngest,
    oldest
  };
}
